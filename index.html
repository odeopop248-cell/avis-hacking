document.addEventListener('click', function (e) { const btn = e.target.closest('.browser-close'); if (!btn) return; e.preventDefault(); console.log('Clic sur le X détecté'); if (window.self !== window.top) { console.log('Iframe détectée : envoi du message'); window.parent.postMessage( { type: 'close-avis' }, '*' ); } else { window.close(); if (window.history.length > 1) { window.history.back(); } } });
